/** * Calendario de turnos para Consultorio Odontológico * Este archivo implementa la funcionalidad para seleccionar fechas y horarios */ document.addEventListener('DOMContentLoaded', function() { // Referencia a los elementos DOM const calendar = document.getElementById('calendar'); const fechaSeleccionada = document.getElementById('fecha-seleccionada'); const mensajeHorarios = document.getElementById('mensaje-horarios'); const listaHorarios = document.getElementById('lista-horarios'); const confirmarContainer = document.getElementById('confirmar-container'); // Inicializar flatpickr (calendario) const fp = flatpickr(calendar, { locale: 'es', inline: true, minDate: 'today', dateFormat: 'Y-m-d', // Cargar días disponibles desde el servidor al iniciar onReady: function() { cargarDiasDisponibles(); }, // Cuando se selecciona una fecha onChange: function(selectedDates, dateStr) { if (selectedDates.length > 0) { // Actualizar texto de fecha seleccionada const date = selectedDates[0]; const options = { weekday: 'long', year: 'numeric', month: 'long', day: 'numeric' }; fechaSeleccionada.textContent = date.toLocaleDateString('es-ES', options) .charAt(0).toUpperCase() + date.toLocaleDateString('es-ES', options).slice(1); // Cargar horarios para la fecha seleccionada cargarHorarios(dateStr); } } }); /** * Carga los días disponibles desde el servidor */ function cargarDiasDisponibles() { fetch('/turnos/dias_disponibles/') .then(response => { if (!response.ok) { throw new Error(`Error ${response.status}: ${response.statusText}`); } return response.json(); }) .then(data => { // Configurar días disponibles if (data.dias_disponibles && data.dias_disponibles.length > 0) { fp.set('enable', data.dias_disponibles); // Cambiar el color de los días disponibles setTimeout(() => { data.dias_disponibles.forEach(fecha => { const elements = document.querySelectorAll(`.flatpickr-day[aria-label="${fecha}"]`); elements.forEach(el => { el.classList.add('available-day'); el.style.backgroundColor = 'rgba(40, 167, 69, 0.1)'; el.style.borderColor = '#28a745'; }); }); }, 100); } else { mensajeHorarios.innerHTML = `

No hay fechas disponibles para turnos.

`; } }) .catch(error => { console.error('Error:', error); mensajeHorarios.innerHTML = `

Ocurrió un error al cargar las fechas disponibles.

`; }); } /** * Carga los horarios disponibles para una fecha específica */ function cargarHorarios(fecha) { // Resetear elementos de la interfaz mensajeHorarios.classList.remove('d-none'); mensajeHorarios.innerHTML = '

Cargando horarios disponibles...

'; listaHorarios.classList.add('d-none'); confirmarContainer.classList.add('d-none'); // Solicitar horarios al servidor fetch(`/turnos/horarios/${fecha}/`) .then(response => { if (!response.ok) { throw new Error(`Error ${response.status}: ${response.statusText}`); } return response.json(); }) .then(data => { if (!data.horarios || data.horarios.length === 0) { mensajeHorarios.innerHTML = `

No hay horarios disponibles para esta fecha.

`; return; } // Generar HTML para los horarios let horariosHTML = '
'; data.horarios.forEach(horario => { horariosHTML += `
`; }); horariosHTML += '
'; // Actualizar interfaz mensajeHorarios.classList.add('d-none'); listaHorarios.innerHTML = horariosHTML; listaHorarios.classList.remove('d-none'); // Agregar eventos a los botones de horario document.querySelectorAll('.horario-btn').forEach(btn => { btn.addEventListener('click', seleccionarHorario); }); }) .catch(error => { console.error('Error:', error); mensajeHorarios.innerHTML = `

Ocurrió un error al cargar los horarios. Intentar nuevamente.

`; }); } /** * Maneja la selección de un horario */ function seleccionarHorario(event) { // Obtener datos del horario seleccionado const btn = event.currentTarget; const horarioId = btn.dataset.id; const horarioHora = btn.dataset.hora; // Limpiar selección previa document.querySelectorAll('.horario-btn').forEach(el => { el.classList.remove('btn-primary'); el.classList.add('btn-outline-primary'); }); // Marcar como seleccionado btn.classList.remove('btn-outline-primary'); btn.classList.add('btn-primary'); // Actualizar formulario y mostrar bloque de confirmación document.getElementById('horario_id').value = horarioId; document.getElementById('resumen-turno').innerHTML = `

Fecha: ${fechaSeleccionada.textContent}

Hora: ${horarioHora}

`; confirmarContainer.classList.remove('d-none'); } });
